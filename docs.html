Payment Gateway API

Dokumentasi integrasi layanan transaksi otomatis.

Base URL: https://payment.warung-api.my.id

Catatan Penting & Panduan Mencegah Kesalahan (Troubleshooting)

  • order_id vs transaction_id: order_id (contoh: 3813354941) hanya digunakan untuk nomor invoice/tampilan struk user. Jangan dipakai untuk mengecek status ke endpoint /check-transaction karena akan selalu menghasilkan status PENDING. Gunakan transaction_id (berawalan pay_...).
  • Status Sukses: Respons status pembayaran yang valid dari server pusat adalah SUCCESS (bukan PAID). Pastikan bot/aplikasi Anda mendeteksi string SUCCESS.
  • Nominal Harga Dinamis: Setiap metode pembayaran memiliki biaya unik masing-masing. Pastikan mengirim parameter price sesuai dengan harga spesifik metode yang dipilih (lihat daftar harga di bawah).

Endpoints

Get Products

GET /api/products

Mendapatkan daftar produk yang tersedia, terbagi menjadi kategori promo dan non-promo beserta struktur harganya.

Request
curl -sS https://payment.warung-api.my.id/api/products
Response
{
  "success": true,
  "data": {
    "products": {
      "promo": [
        {"id": 642, "code": "DANA10K", "name": "10k", "price": 10180, "promo": true}
      ],
      "non_promo": [
        {"id": 12423, "code": "DANABF10", "name": "10K", "price": 10350, "promo": false}
      ]
    }
  }
}

Create Transaction

POST /api/create-transaction

Membuat transaksi baru dan menghasilkan QR code atau data pembayaran.

Request Body
{
  "user_id": "08123456789",
  "product_detail_id": 642,
  "price": 10180,
  "payment_method": "qris",
  "customer_id": "",
  "voucher_code": "",
  "voucher_amount": 0,
  "discounted_price": 0
}
Response
{
  "success": true,
  "data": {
    "transaction_id": "pay_3eum1rJMqu3470",
    "order_id": "3813354941",
    "qr_code_url": "https://...",
    "transaction_at": "2026-09-09 06:57:21"
  }
}

Check Transaction

POST /api/check-transaction

Mengecek status pembayaran transaksi secara berkala (polling) menggunakan transaction_id.

Request Body
{
  "transaction_id": "pay_3eum1rJMqu3470"
}
Response (Pending)
{
  "success": true,
  "data": {
    "status": "PENDING",
    "paid_at": null
  }
}
Response (Success)
{
  "success": true,
  "data": {
    "status": "SUCCESS",
    "paid_at": "2026-09-09 06:57:21"
  }
}

Parameters Reference

Parameter Type Required Description
user_id String Yes Nomor tujuan transaksi (contoh: 08123456789)
product_detail_id Number Yes ID produk referensi dari endpoint /api/products
price Number Yes Nominal harga produk spesifik metode pembayaran
payment_method String No Metode pembayaran (default: qris)
transaction_id String Yes ID unik berawalan pay_... dari create-transaction untuk cek status

Contoh Implementasi Fitur Bot WhatsApp

Modul handler siap pakai yang sudah teruji dan mendeteksi status SUCCESS secara akurat:

const axios = require("axios");

const API_BASE_URL = "https://payment.warung-api.my.id/api";

module.exports = {
  name: "order",
  category: "transaksi",
  commands: ["order", "beli"],
  handle: async ({ text, config, sock, from, fakeQuoted, msg }) => {

    if (!text || text.trim() === "") {
      return await sock.sendMessage(
        from,
        { text: `*Format Salah!*\n\nCara penggunaan:\n.order  \n\nContoh:\n.order DANA10K 08123456789` },
        { quoted: fakeQuoted || msg }
      );
    }

    let args = text.trim().split(/\s+/);
    
    if (args.length > 0 && (args[0].toLowerCase().includes("order") || args[0].toLowerCase().includes("beli"))) {
      args.shift(); 
    }

    if (args.length < 2) {
      return await sock.sendMessage(
        from,
        { text: `*Format Salah!*\nPastikan menyertakan Kode Produk dan Nomor HP.\nContoh: .order DANA10K 08123456789` },
        { quoted: fakeQuoted || msg }
      );
    }

    const kodeProduk = args[0].toUpperCase();
    const nomorHp = args[1];

    try {
      await sock.sendMessage(from, { text: "⏳ Sedang mengecek produk..." }, { quoted: fakeQuoted || msg });
      
      const { data: productResponse } = await axios.get(`${API_BASE_URL}/products`);
      
      const promoList = productResponse.data.products.promo || [];
      const nonPromoList = productResponse.data.products.non_promo || [];
      const allProducts = [...promoList, ...nonPromoList];

      const produk = allProducts.find((p) => p.code === kodeProduk);

      if (!produk) {
        return await sock.sendMessage(
          from,
          { text: `❌ Produk dengan kode *${kodeProduk}* tidak ditemukan.` },
          { quoted: fakeQuoted || msg }
        );
      }

      const payloadTransaction = {
        user_id: nomorHp,
        product_detail_id: produk.id,
        price: produk.price,
        payment_method: "qris",
        customer_id: "",
        voucher_code: "",
        voucher_amount: 0,
        discounted_price: 0
      };

      const { data: trxResponse } = await axios.post(`${API_BASE_URL}/create-transaction`, payloadTransaction);
      
      if (!trxResponse.success) throw new Error("Gagal generate QRIS");
      
      const { transaction_id, order_id, qr_code_url } = trxResponse.data;

      const invoiceText = `🛒 *INVOICE PEMBAYARAN*\n\n` +
                          `• Order ID : ${order_id}\n` +
                          `• Produk : ${produk.name} (${produk.code})\n` +
                          `• Tujuan : ${nomorHp}\n` +
                          `• Total : *Rp ${produk.price.toLocaleString("id-ID")}*\n\n` +
                          `Silakan scan QRIS di atas untuk melakukan pembayaran. Sistem akan otomatis mengecek pembayaranmu dalam 5 menit.`;

      await sock.sendMessage(
        from,
        {
          image: { url: qr_code_url }, 
          caption: invoiceText
        },
        { quoted: fakeQuoted || msg }
      );

      let maxCheck = 30; 
      let isPaid = false;

      while (maxCheck > 0) {
        await new Promise((resolve) => setTimeout(resolve, 10000));

        try {
          // PENTING: Wajib menggunakan transaction_id (pay_...)
          const { data: checkResponse } = await axios.post(`${API_BASE_URL}/check-transaction`, {
            transaction_id: transaction_id 
          });

          // PENTING: Status sukses dari API adalah SUCCESS (bukan PAID)
          if (checkResponse.success && checkResponse.data.status === "SUCCESS") {
            isPaid = true;
            break;
          }
        } catch (err) {
          console.error("Gagal cek status transaksi (retry):", err.message);
        }

        maxCheck--;
      }

      if (isPaid) {
        const successText = `✅ *PEMBAYARAN BERHASIL*\n\nTerima kasih, pembayaran untuk Order ID *${order_id}* telah kami terima.\n\nPesanan *${produk.name}* ke nomor ${nomorHp} sedang diproses!`;
        await sock.sendMessage(from, { text: successText }, { quoted: fakeQuoted || msg });
      } else {
        const timeoutText = `❌ *PEMBAYARAN EXPIRED*\n\nWaktu pembayaran untuk Order ID *${order_id}* telah habis (5 menit). Silakan buat pesanan baru.`;
        await sock.sendMessage(from, { text: timeoutText }, { quoted: fakeQuoted || msg });
      }

    } catch (error) {
      console.error("[Order Error]:", error?.response?.data || error.message);
      return await sock.sendMessage(
        from,
        { text: `❌ Terjadi kesalahan pada server saat memproses pesananmu. Coba beberapa saat lagi.` },
        { quoted: fakeQuoted || msg }
      );
    }
  }
};
